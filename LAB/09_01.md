## CODE
````c++
#include <stdio.h>

int summation(int n) {
    if (n == 1) {
        return 1; 
    } else {
        return n + summation(n - 1); 
    }
}

int main() {
    int n;
    printf("Enter number: ");
    scanf("%d", &n);
    
    if (n < 1) {
        printf("Please enter a number greater than 0.\n");
    } else {
        printf("Summation of 1 to %d is %d\n", n, summation(n));
    }
    
    return 0;
}
````
