````c++
#include <stdio.h>

void hanoi(int m, int i, int j, int k) {
    if (m > 0) {
    
        hanoi(m - 1, i, k, j);
        printf("Move disk %d from tower %d to %d\n", m, i, j);
        hanoi(m - 1, k, j, i);
    }
}

int main() {
    int n;
    printf("Enter number of disks: ");
    scanf("%d", &n);

    printf("Steps to solve Tower of Hanoi with %d disks:\n", n);
    hanoi(n, 1, 3, 2);

    return 0;
}
````
