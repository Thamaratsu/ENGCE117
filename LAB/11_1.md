#include <stdio.h>
#include <stdlib.h>

int *Dijkstra(int *L, int n);

int main() {
    int n = 5, i = 0, j = 0, *d, *g;

    g = (int*)malloc(n * n * sizeof(int));

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            g[i * n + j] = -1;

    g[0 * n + 1] = 100;   g[0 * n + 2] = 80;
    g[0 * n + 3] = 30;    g[0 * n + 4] = 10;
    g[1 * n + 2] = 20;    g[3 * n + 1] = 20;
    g[3 * n + 2] = 20;    g[4 * n + 3] = 10;

    d = Dijkstra(g, n);

    for (i = 0; i < n - 1; i++)
        printf("%d ", d[i]);

    free(g);
    free(d);
    return 0;
}

int *Dijkstra(int *L, int n) {
    int *D = (int*)malloc(n * sizeof(int));
    int *C = (int*)malloc(n * sizeof(int));

    for (int i = 0; i < n; i++) {
        C[i] = (i == 0) ? 0 : 1;
        D[i] = L[0 * n + i];  
    }
    D[0] = 0;

    for (int iter = 0; iter < n - 2; iter++) {

        int v = -1;
        for (int i = 1; i < n; i++) {
            if (C[i] == 1 && D[i] != -1) {
                if (v == -1 || D[i] < D[v])
                    v = i;
            }
        }

        if (v == -1) break;

        C[v] = 0;  

        for (int w = 0; w < n; w++) {
            if (C[w] == 1 && L[v * n + w] != -1) {
                int newDist = D[v] + L[v * n + w];
                if (D[w] == -1 || newDist < D[w])
                    D[w] = newDist;
            }
        }
    }

    int *result = (int*)malloc((n - 1) * sizeof(int));
    for (int i = 0; i < n - 1; i++)
        result[i] = D[i + 1];

    free(D);
    free(C);
    return result;
}
