#include <stdio.h>
#include <stdlib.h>

int *KnapsackGreedy(int *w, int *v, int n, int wx);

int main() {
    int n = 5, wx = 11;
    int w[5] = { 1, 2, 5, 6, 7 };
    int v[5] = { 1, 6, 18, 22, 28 };
    int *x = KnapsackGreedy(w, v, n, wx);
    for (int i = 0; i < n; i++) printf("%d ", x[i]);
    free(x);
    return 0;
}

int *KnapsackGreedy(int *w, int *v, int n, int wx) {
    int *x = (int*)malloc(n * sizeof(int));
    int *y = (int*)malloc(n * sizeof(int));

    for (int i = 0; i < n; i++) {
        x[i] = 0;
        y[i] = 1; 
    }

    int weight = 0;
    int yCount = n;

    while (weight < wx && yCount != 0) {

        int best = -1;
        for (int i = 0; i < n; i++) {
            if (y[i] == 1) {
                if (best == -1 || (v[i] * w[best] > v[best] * w[i]))
                    best = i;
            }
        }

        if (weight + w[best] <= wx) {
            x[best] = 1;
            weight = weight + w[best];
        }

        y[best] = 0;  
        yCount--;
    }

    free(y);
    return x;
}
