#include <stdio.h>

#include <string.h>

int *KnapsackDP(int *w, int *v, int n, int wx);

int main() {
    int n = 5, wx = 11;
    int w[5] = { 1, 2, 5, 6, 7 };
    int v[5] = { 1, 6, 18, 22, 28 };
    int *x;
    x = KnapsackDP(w, v, n, wx);
    for (int i = 0; i < n; i++) printf("%d ", x[i]);
    free(x);
    return 0;
}

int *KnapsackDP(int *w, int *v, int n, int wx) {

    int **u = (int**)malloc(n * sizeof(int*));
    int **y = (int**)malloc(n * sizeof(int*));
    for (int i = 0; i < n; i++) {
        u[i] = (int*)malloc((wx + 1) * sizeof(int));
        y[i] = (int*)calloc((wx + 1), sizeof(int));
    }

    for (int i = 0; i < n; i++) {
        for (int j = 0; j <= wx; j++) {

            int a, Sa;
            if (i - 1 < 0) { a = 0;           Sa = 0; }
            else            { a = u[i-1][j];   Sa = y[i-1][j]; }

            int b, Sb;
            if (j - w[i] < 0) {
                b = -1;   
                Sb = 0;
            } else if (j - w[i] == 0 || i - 1 < 0) {
                b  = v[i];
                Sb = (1 << i);
            } else {
                b  = u[i-1][j - w[i]] + v[i];
                Sb = (1 << i) | y[i-1][j - w[i]];
            }

            if (a > b) { u[i][j] = a;  y[i][j] = Sa; }
            else        { u[i][j] = b;  y[i][j] = Sb; }
        }
    }

    int *x = (int*)calloc(n, sizeof(int));
    int chosen = y[n-1][wx];
    for (int i = 0; i < n; i++) {
        if (chosen & (1 << i)) x[i] = 1;
    }

    for (int i = 0; i < n; i++) { free(u[i]); free(y[i]); }
    free(u); free(y);
    return x;
}
